using osu.Framework.Localisation;

namespace TestProject.Localisation
{
    public static class ProgramStrings
    {
        private const string prefix = @"TestProject.Localisation.Program";

        /// <summary>
        /// "{0}a{1}{2}|{0}b{1}{2}"
        /// </summary>
        public static LocalisableString AB(string stringParam, int quantityParam, int otherParam) => new PluralisableString(new TranslatableString(getKey(@"ab"), @"{0}a{1}{2}|{0}b{1}{2}", stringParam, quantityParam, otherParam), quantityParam, '|');

        private static string getKey(string key) => $@"{prefix}:{key}";
    }
}
